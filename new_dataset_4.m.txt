%% FIR Filter + Dataset with Tunable Windows + Main Lobe + Side Lobes
clear; clc; close all;

% -------------------------------------------------------------------------
% 1. Filter Design Parameters (NOW ALL VARY)
% -------------------------------------------------------------------------
cutoffs = 15010:10:20000;
filter_lengths = 21:5:221;

% --- Base Windows ---
base_windows = {...
'Rectangular', 'Hann', 'Triangular', 'Blackman', ...
'Tukey alpha=0.5', 'Bartlett', 'Hamming', 'Bohman', ...
'Flattop', 'Blackman-harris','Parzen'};

% --- Tunable Windows ---
kaiser_betas = [0, 3, 5, 7, 9];
gaussian_alphas = [1.5, 2.5, 3.5, 4.5];
cheb_dBs = [40, 60, 80, 100];

% --- NEW: Taylor Parameters ---
taylor_nbars = 3:6;
taylor_slls = 25:5:50;   % 25,30,35,40,45,50 dB

% --- Build full list ---
window_names = base_windows;

for b = kaiser_betas
    window_names{end+1} = sprintf('Kaiser beta=%.1f', b);
end

for a = gaussian_alphas
    window_names{end+1} = sprintf('Gaussian alpha=%.1f', a);
end

for d = cheb_dBs
    window_names{end+1} = sprintf('Chebyshev %d dB', d);
end

% --- Add Taylor Combinations ---
for nb = taylor_nbars
    for s = taylor_slls
        window_names{end+1} = sprintf('Taylor nbar=%d sll=%d', nb, s);
    end
end

num_windows = length(window_names);

% --- Pre-allocate dataset ---
num_cutoffs = length(cutoffs);
num_lengths = length(filter_lengths);
params_all = cell(num_cutoffs * num_lengths * num_windows, 10);
dataset_idx = 1;

% -------------------------------------------------------------------------
% MAIN LOOP
% -------------------------------------------------------------------------
for c = 1:num_cutoffs
    Fc = cutoffs(c);
    for l = 1:num_lengths
        N = filter_lengths(l);
        
        Fs = ceil(Fc * 4 ) ;
        if Fc >= Fs/2, continue; end
        
        Wn = Fc / (Fs/2);
        
        h_all = cell(1, num_windows);
        
        for w = 1:num_windows
            name = window_names{w};
            win = [];
            
            if contains(name, 'Rectangular'), win = rectwin(N);
            elseif contains(name, 'Hann'), win = hann(N);
            elseif contains(name, 'Triangular'), win = triang(N);
            elseif contains(name, 'Blackman'), win = blackman(N);
            elseif contains(name, 'Tukey'), win = tukeywin(N, 0.5);
            elseif contains(name, 'Bartlett'), win = bartlett(N);
            elseif contains(name, 'Hamming'), win = hamming(N);
            elseif contains(name, 'Bohman'), win = bohmanwin(N);
            elseif contains(name, 'Flattop'), win = flattopwin(N);
            elseif contains(name, 'Blackman-harris'), win = blackmanharris(N);
            elseif contains(name, 'Parzen'), win = parzenwin(N);
            
            elseif contains(name, 'Kaiser')
                beta = str2double(regexp(name, '\d+\.?\d*', 'match', 'once'));
                win = kaiser(N, beta);
                
            elseif contains(name, 'Gaussian')
                alpha = str2double(regexp(name, '\d+\.?\d*', 'match', 'once'));
                win = gausswin(N, alpha);
                
            elseif contains(name, 'Chebyshev')
                dB = str2double(regexp(name, '\d+', 'match', 'once'));
                win = chebwin(N, dB);
                
            elseif contains(name, 'Taylor')
                tokens = regexp(name, '\d+', 'match');
                nbar = str2double(tokens{1});
                sll = -str2double(tokens{2});  % MATLAB expects negative
                win = taylorwin(N, nbar, sll);
            end
            
            if ~isempty(win)
                h = fir1(N-1, Wn, 'low', win);
                h_all{w} = h;
            end
        end
        
        % -----------------------------------------------------------------
        % Noisy Sine Wave
        % -----------------------------------------------------------------
        duration = 1.0;
        t = 0:1/Fs:duration-1/Fs;
        f_sine = Fc / 2;
        x_clean = sin(2*pi*f_sine*t);
        
        SNR_in_dB = 10;
        P_sig = mean(x_clean.^2);
        P_noise = P_sig / (10^(SNR_in_dB/10));
        noise = sqrt(P_noise) * randn(size(t));
        x_noisy = x_clean + noise;
        
        % -----------------------------------------------------------------
        % Compute Parameters
        % -----------------------------------------------------------------
        for w = 1:num_windows
            h = h_all{w};
            if isempty(h), continue; end
            
            [H, f] = freqz(h, 1, 2^14, Fs);
            Hmag_dB = 20*log10(abs(H) + eps);
            
            pass_ripple = max(Hmag_dB(f <= Fc)) - min(Hmag_dB(f <= Fc));
            
            stopResp = Hmag_dB(f >= Fc);
            [pks, locs] = findpeaks(stopResp, 'SortStr', 'descend');
            
            if ~isempty(pks)
                stop_atten = -pks(1);
                main_lobe_dB = pks(1);
            else
                stop_atten = NaN;
                main_lobe_dB = NaN;
            end
            
            side_lobes = zeros(1,3);
            side_lobes_str = 'NaN dB, NaN dB, NaN dB';
            
            if length(pks) >= 4
                side_lobes = pks(2:4);
                side_lobes_str = sprintf('%.2f dB, %.2f dB, %.2f dB', side_lobes);
            elseif length(pks) >= 2
                side_lobes(1:length(pks)-1) = pks(2:end);
                side_lobes_str = sprintf('%.2f dB, %.2f dB, %.2f dB', side_lobes);
            end
            
            idx3dB = find(Hmag_dB <= -3, 1, 'first');
            if ~isempty(idx3dB)
                f3dB = f(idx3dB);
            else
                f3dB = NaN;
            end
            
            if ~isempty(locs)
                stopIdx = find(f >= Fc, 1, 'first');
                fFirst = f(stopIdx + locs(1) - 1);
            else
                fFirst = Fs/2;
            end
            
            trans_width = fFirst - f3dB;
            
            y_clean_filt = filter(h, 1, x_clean);
            y_noisy_filt = filter(h, 1, x_noisy);
            residual = y_noisy_filt - y_clean_filt;
            snr_out = 10*log10(var(y_clean_filt)/(var(residual)+eps));
            
            params_all(dataset_idx, :) = ...
                {Fc, Fs, N, window_names{w}, pass_ripple, ...
                stop_atten, trans_width, main_lobe_dB, ...
                side_lobes_str, snr_out};
            
            dataset_idx = dataset_idx + 1;
        end
    end
end

params_all(dataset_idx:end, :) = [];

if exist('filter_parameters.csv', 'file')
    delete('filter_parameters.csv');
end

colNames = {'cutoff','sampling frequency','filter length','window',...
'passband ripples','stop band attenuation','transition band width',...
'main_lobe_dB','side_lobes','snr'};

T = cell2table(params_all, 'VariableNames', colNames);
writetable(T, 'filter_parameters.csv');

disp('SUCCESS: DATASET WITH EXTENDED TAYLOR WINDOWS!');
fprintf(' Total Rows: %d\n', height(T));
